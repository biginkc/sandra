       ```ts
       export function computeDrain(args: {
         total: number;
         paceSeconds: number;
         dailyCap: number | undefined;
         now: Date;
       }): {
         perDay: { dayLabel: string; count: number }[];
         lastSendLocal: string | null;
         pastCutoffCount: number;  // # that would fall past 9 PM PT (mirrors federal TCPA)
       } {
         const { total, paceSeconds, dailyCap, now } = args;
         if (total === 0 || !Number.isFinite(paceSeconds) || paceSeconds <= 0) {
           return { perDay: [], lastSendLocal: null, pastCutoffCount: 0 };
         }
         const perDay: { dayLabel: string; count: number }[] = [];
         let remaining = total;
         let dayStart = now.getTime();
         let pastCutoffCount = 0;
         let lastSendMs = dayStart;
         const dayLabel = (ms: number, idx: number) => {
           if (idx === 0) return "Today";
           if (idx === 1) return "Tomorrow";
           return new Intl.DateTimeFormat("en-US", { weekday: "short", timeZone: "America/Los_Angeles" }).format(new Date(ms));
         };
         let dayIdx = 0;
         while (remaining > 0) {
           const cap = dailyCap ?? remaining;
           const inThisDay = Math.min(remaining, cap);
           perDay.push({ dayLabel: dayLabel(dayStart, dayIdx), count: inThisDay });
           // Last send in this day = dayStart + (inThisDay - 1) * paceSeconds * 1000
           lastSendMs = dayStart + Math.max(0, inThisDay - 1) * paceSeconds * 1000;
           // Count any that would land past 9 PM PT same calendar day (federal TCPA cutoff).
           // The server defers them on release; we surface the count so the UI shows the rollover.
           const ptNinePm = (() => {
             const d = new Date(dayStart);
             const fmt = new Intl.DateTimeFormat("en-US", {
               timeZone: "America/Los_Angeles",
               year: "numeric", month: "2-digit", day: "2-digit",
             });
             const parts = fmt.formatToParts(d);
             const y = Number(parts.find(p => p.type === "year")!.value);
             const m = Number(parts.find(p => p.type === "month")!.value);
             const day = Number(parts.find(p => p.type === "day")!.value);
             return Date.UTC(y, m - 1, day, 21 + 8, 0, 0); // 21:00 PT ≈ 05:00 UTC next day at -08:00
           })();
           if (lastSendMs > ptNinePm) {
             const overflowStart = Math.max(0, Math.ceil((ptNinePm - dayStart) / (paceSeconds * 1000)));
             pastCutoffCount += Math.max(0, inThisDay - overflowStart);
           }
           remaining -= inThisDay;
           dayIdx += 1;
           // Next day starts at 8 AM PT
           const tomorrow = new Date(dayStart);
           tomorrow.setUTCDate(tomorrow.getUTCDate() + 1);
           const fmt = new Intl.DateTimeFormat("en-US", {
             timeZone: "America/Los_Angeles",
             year: "numeric", month: "2-digit", day: "2-digit",
           });
           const parts = fmt.formatToParts(tomorrow);
           const y = Number(parts.find(p => p.type === "year")!.value);
           const m = Number(parts.find(p => p.type === "month")!.value);
           const day = Number(parts.find(p => p.type === "day")!.value);
           dayStart = Date.UTC(y, m - 1, day, 16, 0, 0); // 08:00 PT == 16:00 UTC
         }
         const lastSendLocal = new Intl.DateTimeFormat("en-US", {
           timeZone: "America/Los_Angeles",
           hour: "numeric", minute: "2-digit", weekday: "short",
         }).format(new Date(lastSendMs));
         return { perDay, lastSendLocal, pastCutoffCount };
       }
       ```
