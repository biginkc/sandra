export interface DispatchCalendarTaskInput {
  taskId: string;
  assigneeId: string;          // Sandra user_id
  taskTitle: string;
  propertyAddress: string;
  dueAt: string;               // ISO timestamptz
  timezone: string;            // IANA tz; loadIntegrationPrefs default = "America/Chicago"
  deepLink: string;            // https://sandra-sooty.vercel.app/messages?property_id=...
}

export interface DispatchCalendarResult {
  inserted: boolean;
  eventId?: string;
  reason?: "no_token" | "pref_disabled" | "error";
}

// Insert on task create. Best-effort. Never throws.
export async function dispatchTaskCalendarEvent(
  input: DispatchCalendarTaskInput,
): Promise<DispatchCalendarResult>;

// Update on due_at change. Reads tasks.google_calendar_event_id; falls back to insert if column is null.
// Best-effort. Never throws.
export async function dispatchTaskCalendarEventUpdate(
  input: DispatchCalendarTaskInput,
): Promise<DispatchCalendarResult>;

<gotchas>
- Pitfall 3: tokens.expiry_date is in MILLISECONDS (already converted at OAuth callback time per Plan 06). Store as ISO; the SDK's setCredentials accepts ms via expiry_date.
- Pitfall 4: oauth2Client.on('tokens', ...) fires when the SDK auto-refreshes. If newTokens.refresh_token is undefined, DO NOT clobber the stored refresh_token — pass null and let upsert_oauth_token COALESCE preserve it.
- The googleapis SDK lazily refreshes the access_token on the first 401. Don't try to pre-emptively refresh — let the SDK handle it.
- Calendar event reminders: do NOT override `reminders` — inherit the user's default reminders. CONTEXT.md doesn't specify and changing this surprises users.
- Time math: start = new Date(dueAt); end = new Date(start.getTime() + 30 * 60 * 1000). 30 minutes per D-06.
- D-07: task completion is a NO-OP. There is no Delete function in this plan. If a future V3 changes that policy, add a separate plan.
- Insert vs update distinction: Plan 10's wire-up calls dispatchTaskCalendarEvent on task CREATE; dispatchTaskCalendarEventUpdate on due_at CHANGE. The update function reads tasks.google_calendar_event_id; if null (first time), it falls through to dispatchTaskCalendarEvent.
- `calendar.events.insert({ calendarId: "primary", requestBody: {...} })` — primary is a literal string per Google docs.
- `calendar.events.update({ calendarId: "primary", eventId, requestBody: {...} })` — replaces the event entirely. Use `calendar.events.patch` for partial updates if we ever need to. For V2, `update` (full replace) is fine because we always know the canonical state.
</gotchas>
