    ```typescript
    import { redirect } from "next/navigation";

    import { Page } from "@/components/page";
    import { PageHeader } from "@/components/page-header";
    import { createClient } from "@/lib/supabase/server";

    import { getIntegrationStatus } from "./actions";
    import { IntegrationsForm } from "./form";

    interface Props {
      searchParams: Promise<{ connected?: string; error?: string }>;
    }

    export default async function IntegrationsSettingsPage(props: Props) {
      const supabase = await createClient();
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) redirect("/login");

      const status = await getIntegrationStatus();
      const params = await props.searchParams;

      const banner = params.connected
        ? { variant: "success" as const, text: `${params.connected === "slack" ? "Slack" : "Google Calendar"} connected.` }
        : params.error
          ? { variant: "error" as const, text: bannerForError(params.error) }
          : null;

      return (
        <Page>
          <PageHeader
            breadcrumb={[{ label: "Settings" }, { label: "Integrations" }]}
            title="Integrations"
            description="Connect Slack and Google Calendar to receive task notifications outside Sandra. One-way outbound — Sandra remains the source of truth."
          />
          {banner && (
            <div role="status" data-banner-variant={banner.variant} className="...">
              {banner.text}
            </div>
          )}
          {!status.ok ? (
            <div className="text-destructive text-sm">
              Failed to load integration status: {status.error.message}
            </div>
          ) : (
            <IntegrationsForm initial={status.data} />
          )}
        </Page>
      );
    }

    function bannerForError(code: string): string {
      switch (code) {
        case "state": return "Connection failed: state mismatch. Please try again.";
        case "callback": return "Connection failed during the callback. Please try again.";
        case "config": return "Server is missing OAuth credentials. Contact the admin.";
        case "start": return "Could not start the connection. Please try again.";
        default: return `Connection error: ${code}`;
      }
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Server actions | User-authenticated; RLS protects user_integration_prefs writes; service-role only for token deletes (gated behind disconnectIntegration's auth check). |
| Server action → Slack/Google revoke endpoints | Best-effort; failure logged but does not block local row deletion. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-09-01 | Spoofing | unauthenticated user calls disconnectIntegration | mitigate | Each server action begins with `auth.getUser()`; returns err{code:"AUTH"} if no user. |
| T-04-09-02 | Tampering | user disconnects another user's tokens | mitigate | deleteOAuthTokens uses the authenticated user.id from session — no userId parameter accepted from the client. |
| T-04-09-03 | Information Disclosure | toggle/timezone leak via revalidate | accept | revalidatePath("/settings/integrations") only re-renders the user's own page; data is per-user. |
| T-04-09-04 | Tampering | timezone arbitrary string | mitigate | ALLOWED_TIMEZONES allowlist; setTimezoneAction returns VALIDATION error on any value not in the list. |
| T-04-09-05 | Information Disclosure | banner reflects user-controlled query string | mitigate | bannerForError uses a switch over fixed codes; unknown codes render the code verbatim but not as HTML — React escapes by default. |
| T-04-09-06 | Repudiation | who disconnected? | accept | Existing audit (Sentry/structured logs) captures user.id at the auth gate. Sufficient for V2. |
| T-04-09-07 | Denial of Service | upstream revoke timeout | mitigate | Best-effort revoke wrapped in try/catch + reportError; the local delete proceeds regardless. |
</threat_model>
