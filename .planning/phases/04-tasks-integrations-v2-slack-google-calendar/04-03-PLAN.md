    ```typescript
    import type { SupabaseClient } from "@supabase/supabase-js";

    import { DatabaseError } from "@/lib/errors/classes";
    import { reportError } from "@/lib/errors/report";
    import type { Database } from "@/lib/supabase/types";

    export interface IntegrationPrefs {
      slackEnabled: boolean;
      calendarEnabled: boolean;
      timezone: string;
    }

    export type IntegrationChannel = "slack" | "google_calendar";

    const DEFAULT_PREFS: IntegrationPrefs = {
      slackEnabled: true,
      calendarEnabled: true,
      timezone: "America/Chicago",
    };

    export async function loadIntegrationPrefs(
      supabase: SupabaseClient<Database>,
      userId: string,
    ): Promise<IntegrationPrefs> {
      try {
        const { data, error } = await supabase
          .from("user_integration_prefs")
          .select("channel, enabled, timezone")
          .eq("user_id", userId);
        if (error) {
          reportError(error, { tags: { surface: "load_integration_prefs" }, extra: { userId } });
          return DEFAULT_PREFS;
        }
        const slackRow = data?.find((r) => r.channel === "slack");
        const calRow = data?.find((r) => r.channel === "google_calendar");
        return {
          slackEnabled: slackRow?.enabled ?? true,
          calendarEnabled: calRow?.enabled ?? true,
          timezone: calRow?.timezone ?? slackRow?.timezone ?? "America/Chicago",
        };
      } catch (e) {
        reportError(e, { tags: { surface: "load_integration_prefs" }, extra: { userId } });
        return DEFAULT_PREFS;
      }
    }

    export async function setChannelEnabled(
      supabase: SupabaseClient<Database>,
      userId: string,
      channel: IntegrationChannel,
      enabled: boolean,
    ): Promise<void> {
      const { error } = await supabase
        .from("user_integration_prefs")
        .upsert(
          { user_id: userId, channel, enabled, updated_at: new Date().toISOString() },
          { onConflict: "user_id,channel" },
        );
      if (error) throw new DatabaseError(`setChannelEnabled failed: ${error.message}`);
    }

    export async function setTimezone(
      supabase: SupabaseClient<Database>,
      userId: string,
      timezone: string,
    ): Promise<void> {
      const rows: IntegrationChannel[] = ["slack", "google_calendar"];
      const updated_at = new Date().toISOString();
      const { error } = await supabase
        .from("user_integration_prefs")
        .upsert(
          rows.map((channel) => ({ user_id: userId, channel, timezone, updated_at })),
          { onConflict: "user_id,channel" },
        );
      if (error) throw new DatabaseError(`setTimezone failed: ${error.message}`);
    }
    ```

    Note: `setChannelEnabled` and `setTimezone` rely on the row being insertable with default `enabled=true` (CHECK + DEFAULT in the migration). Upsert with only `enabled` set works because the timezone has a NOT NULL DEFAULT.
