    ```typescript
    import { createHmac, timingSafeEqual } from "node:crypto";

    const DEFAULT_MAX_AGE_SEC = 600;

    export function signOAuthState(opts: { userId: string; secret: string; now?: number }): string {
      const now = opts.now ?? Math.floor(Date.now() / 1000);
      const payload = `${opts.userId}.${now}`;
      const hmac = createHmac("sha256", opts.secret).update(payload).digest("hex");
      return `${payload}.${hmac}`;
    }

    export function verifyOAuthState(opts: {
      state: string;
      secret: string;
      expectedUserId: string;
      now?: number;
      maxAgeSec?: number;
    }): boolean {
      const parts = opts.state.split(".");
      if (parts.length !== 3) return false;
      const [userId, tsStr, hmacHex] = parts;
      if (userId !== opts.expectedUserId) return false;
      const ts = Number(tsStr);
      if (!Number.isFinite(ts)) return false;
      const now = opts.now ?? Math.floor(Date.now() / 1000);
      const maxAge = opts.maxAgeSec ?? DEFAULT_MAX_AGE_SEC;
      if (now - ts > maxAge || ts > now + 60) return false;     // future-dated up to 60s tolerance
      const expected = createHmac("sha256", opts.secret).update(`${userId}.${tsStr}`).digest("hex");
      const a = Buffer.from(hmacHex, "hex");
      const b = Buffer.from(expected, "hex");
      if (a.length !== b.length || a.length === 0) return false;
      return timingSafeEqual(a, b);
    }
    ```

       ```typescript
         try {
           const supabase = await createClient();
           const { data: { user } } = await supabase.auth.getUser();
       ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-05-01 | Spoofing | OAuth CSRF (attacker-initiated authorize) | mitigate | Signed state token: `${userId}.${ts}.${hmac}`. Callback verifies HMAC AND that userId matches the current session AND timestamp is within 10 min. Reject otherwise. |
| T-04-05-02 | Spoofing | code interception | mitigate | TLS only; redirect_uri pinned in Slack app config to https://sandra-sooty.vercel.app/api/oauth/slack/callback. Code is single-use per Slack. |
| T-04-05-03 | Tampering | state token forgery | mitigate | timingSafeEqual byte compare on HMAC; reject malformed (non-3-segment) state. |
| T-04-05-04 | Replay | stale state token | mitigate | maxAgeSec=600 on verifyOAuthState. After 10 min the click is dead. |
| T-04-05-05 | Information Disclosure | client_secret in browser | mitigate | client_secret never sent to browser. Only client_id appears in the redirect URL — that's safe by Slack's design. |
| T-04-05-06 | Information Disclosure | tokens in logs | mitigate | The callback never logs tokens. reportError is called only on errors and the catch block does not include tokens in extras.
