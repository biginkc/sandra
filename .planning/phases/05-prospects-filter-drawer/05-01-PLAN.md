```ts
// 19 block-kind literals — names match SPEC.md §41-46 exactly
export const BLOCK_KINDS = [
  // General
  "list", "tag", "list_count", "vacancy", "cass", "outreach_dispo", "source",
  // Property
  "beds", "baths", "year_built", "state", "market",
  // Owner
  "absentee",
  // Value & Equity
  "estimated_value", "equity_pct",
  // Status & Engagement
  "pipeline_status", "engagement", "assignee", "created_date",
  // Schema-audit additions
  "has_unread_inbound", "needs_human_attention", "has_open_tasks", "motivation_level",
] as const;
export type BlockKind = (typeof BLOCK_KINDS)[number];

// Tri-state combinator for multi-select blocks
export type Combinator = "all" | "any" | "not";  // "all" = AND, "any" = OR, "not" = NOT IN

// Tri-state boolean for boolean blocks (Vacancy, Absentee, Has Unread Inbound, Needs Human Attention)
export type TriBool = "any" | "yes" | "no";  // "any" = no predicate; "no" = false OR null

// Date mode (Created Date)
export type DateMode =
  | { mode: "fixed"; from: string | null; to: string | null }    // ISO dates
  | { mode: "since"; days: number }                              // rolling window
  | { mode: "prior"; days: number };                             // rolling window

// Numeric range (List Count, Beds, Baths, Year Built, Estimated Value, Equity %)
export type NumRange = { min: number | null; max: number | null };

// Per-kind config (one variant per kind in the discriminated union)
export type FilterBlock =
  | { id: string; kind: "list";              combinator: Combinator; values: string[] }            // list ids
  | { id: string; kind: "tag";               combinator: Combinator; values: string[] }            // tag ids (custom only)
  | { id: string; kind: "list_count";        range: NumRange }
  | { id: string; kind: "vacancy";           tri: TriBool }
  | { id: string; kind: "cass";              combinator: Combinator; values: string[] }            // verified|unverified|invalid|ambiguous
  | { id: string; kind: "outreach_dispo";    combinator: Combinator; values: string[] }
  | { id: string; kind: "source";            combinator: Combinator; values: string[] }            // properties.source enum
  | { id: string; kind: "beds";              range: NumRange }
  | { id: string; kind: "baths";             range: NumRange }
  | { id: string; kind: "year_built";        range: NumRange }
  | { id: string; kind: "state";             combinator: Combinator; values: string[] }            // 2-letter
  | { id: string; kind: "market";            combinator: Combinator; values: string[] }            // counties.market
  | { id: string; kind: "absentee";          tri: TriBool }
  | { id: string; kind: "estimated_value";   range: NumRange }
  | { id: string; kind: "equity_pct";        range: NumRange }
  | { id: string; kind: "pipeline_status";   combinator: Combinator; values: string[] }            // properties.status enum
  | { id: string; kind: "engagement";        combinator: Combinator; values: ("never_contacted" | "attempted" | "replied" | "opted_out")[] }
  | { id: string; kind: "assignee";          combinator: Combinator; values: string[] }            // user ids; "unassigned" sentinel allowed
  | { id: string; kind: "created_date";      date: DateMode }
  | { id: string; kind: "has_unread_inbound";       tri: TriBool }
  | { id: string; kind: "needs_human_attention";    tri: TriBool }
  | { id: string; kind: "has_open_tasks";           tri: TriBool }
  | { id: string; kind: "motivation_level";  combinator: Combinator; values: string[] };           // properties.motivation_level enum

export type BlockStack = FilterBlock[];

export type FilterState = { v: 1; blocks: BlockStack };

export const EMPTY_FILTER_STATE: FilterState = { v: 1, blocks: [] };

export function newBlockId(): string;             // wraps crypto.randomUUID()
export function encodeFilters(s: FilterState): string;   // returns the value for the ?filters= URL param (already URL-encoded)
export function decodeFilters(raw: string | null | undefined): FilterState;  // fail-closed on malformed
```

Other plans depend on this exact API surface. If you need to deviate, document the change in this plan's SUMMARY so downstream plans can adjust.

    Implementation notes:
    1. `BLOCK_KINDS` is a `const` tuple — derive `BlockKind` via `(typeof BLOCK_KINDS)[number]`.
    2. `FilterBlock` is a discriminated union; each variant has the `id: string` + `kind: <literal>` discriminant. Keep variants in the same order as `BLOCK_KINDS` for human readability.

       - Return the encoded string.
    5. `decodeFilters(raw)`:
       - If `raw == null` or empty string → return `EMPTY_FILTER_STATE`.
       - Try `JSON.parse(decodeURIComponent(raw))`. On any throw → return `EMPTY_FILTER_STATE`.
       - If `parsed.v !== 1` → return `EMPTY_FILTER_STATE`.
       - Filter `parsed.blocks` to keep only entries where:
         a) `block.id` is a string
         b) `block.kind` is in `BLOCK_KINDS`
         c) (lightweight per-kind validation: matching shape exists — for combinator blocks, `combinator ∈ ["all","any","not"]` else coerce to `"all"`; for tri-bool, `tri ∈ ["any","yes","no"]` else coerce to `"any"`; for `range`, both fields are number-or-null else coerce to `null`; for `created_date`, `date.mode ∈ ["fixed","since","prior"]`).
       - Return `{ v: 1, blocks: validBlocks }`.
    6. Add a JSDoc on the file explaining the version-bump policy: bumping `v` is a breaking change; old URLs decode to EMPTY_FILTER_STATE and the back-compat shim in `prospects-query.ts` re-translates from old chip params.
