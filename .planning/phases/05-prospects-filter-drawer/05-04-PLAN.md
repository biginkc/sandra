    ## Across-block semantics (AND)
    - Chaining the builder produces AND. `applyFilters(builder, [A, B])` ≡ `applyBlock(applyBlock(builder, A), B)`.

    ## Within multi-select block (combinator)
    - `combinator: "all"` (AND across selected) — for list/tag, use `.in("id", propertyIds)` where propertyIds = INTERSECTION of property-list members per value (multi-step). For other multi-selects with single column, "all" + multiple values is equivalent to "any" because a single column can only have one value at a time — DOCUMENT this in the code comment for kinds where it collapses.
    - `combinator: "any"` (OR across selected) — `.in("col", values)`.
    - `combinator: "not"` (NOT IN) — `.not("col", "in", "(val1,val2)")` formatted per Supabase syntax. For list/tag: filter properties whose stack-counts INTERSECT empty with the value set.

    ## Tri-state booleans (TriBool)
    - `tri: "any"` — no predicate added.
    - `tri: "yes"` — `.eq("col", true)`.
    - `tri: "no"` — `.or("col.eq.false,col.is.null")` so NULL counts as false.

    ## Numeric range (NumRange)
    - `min: null, max: null` → no predicate.
    - `min: x, max: null` → `.gte("col", x)`.
    - `min: null, max: y` → `.lte("col", y)`.
    - `min: x, max: y` → `.gte("col", x).lte("col", y)`.

    ## Date mode (DateMode)
    - `mode: "fixed"` with `from` and/or `to` ISO strings — `.gte`/`.lte` on `created_at`.
    - `mode: "since", days: N` — `.gte("created_at", new Date(now - N*86400000).toISOString())`. Re-evaluates on each render (rolling).
    - `mode: "prior", days: N` — `.lte("created_at", new Date(now - N*86400000).toISOString())`.

     * does not add it.
     */
    export async function applyBlock(
      builder: ProspectsBuilder,
      block: FilterBlock,
      sb: SbClient,
    ): Promise<ProspectsBuilder> {
      switch (block.kind) {
        case "vacancy": return applyTriBool(builder, "is_vacant", block.tri);
        case "cass":    return applyMultiSelect(builder, "cass_status", block.combinator, block.values);
        case "list":    return await applyListBlock(builder, block, sb);
        case "list_count": return await applyListCountBlock(builder, block, sb);
        case "equity_pct": return applyNumRange(builder, "equity_pct", block.range);
        // ... all 23 kinds
        default:
          // exhaustiveness check
          const _exhaustive: never = block;
          return builder;
      }
    }

    export async function applyFilters(
      builder: ProspectsBuilder,
      blocks: BlockStack,
      sb: SbClient,
    ): Promise<ProspectsBuilder> {
      let b = builder;
      for (const block of blocks) {
        b = await applyBlock(b, block, sb);
      }
      return b;
    }

    // ---- helpers ----

    function applyTriBool(builder: ProspectsBuilder, col: string, tri: "any" | "yes" | "no"): ProspectsBuilder {
      if (tri === "any") return builder;
      if (tri === "yes") return builder.eq(col, true);
      return builder.or(`${col}.eq.false,${col}.is.null`);
    }

    function applyMultiSelect(
      builder: ProspectsBuilder,
      col: string,
      combinator: "all" | "any" | "not",
      values: string[],
    ): ProspectsBuilder {
      if (values.length === 0) return builder;
      // Document: for single-column kinds, "all" collapses to "any" because a row
      // can only have one value at a time. We honor "all" semantically by treating
      // it like "any" here. List/Tag override this via property junctions.
      if (combinator === "not") {
        return builder.not(col, "in", `(${values.map(v => `"${v}"`).join(",")})`);
      }
      return builder.in(col, values);
    }

    function applyNumRange(builder: ProspectsBuilder, col: string, range: { min: number | null; max: number | null }): ProspectsBuilder {
      let b = builder;
      if (range.min != null) b = b.gte(col, range.min);
      if (range.max != null) b = b.lte(col, range.max);
      return b;
    }

    // applyListBlock: pre-fetch property_lists rows; intersect for "all"; apply .in("id", ids)
    async function applyListBlock(builder: ProspectsBuilder, block: Extract<FilterBlock, {kind:"list"}>, sb: SbClient): Promise<ProspectsBuilder> {
      if (block.values.length === 0) return builder;
      const { data } = await sb.from("property_lists").select("property_id, list_id").in("list_id", block.values);
      const byProp: Map<string, Set<string>> = new Map();
      for (const r of data ?? []) {
        if (!byProp.has(r.property_id)) byProp.set(r.property_id, new Set());
        byProp.get(r.property_id)!.add(r.list_id);
      }
      let propIds: string[] = [];
      if (block.combinator === "all") {
        for (const [pid, set] of byProp) if (block.values.every(v => set.has(v))) propIds.push(pid);
        return propIds.length ? builder.in("id", propIds) : builder.in("id", ["__no_match__"]);
      }
      if (block.combinator === "not") {
        for (const [pid] of byProp) propIds.push(pid);
        return propIds.length ? builder.not("id", "in", `(${propIds.map(id=>`"${id}"`).join(",")})`) : builder;
      }
      // "any"
      for (const [pid] of byProp) propIds.push(pid);
      return propIds.length ? builder.in("id", propIds) : builder.in("id", ["__no_match__"]);
    }

    // applyListCountBlock: pre-fetch property_stack_counts; apply .in("id", ids)
    async function applyListCountBlock(builder: ProspectsBuilder, block: Extract<FilterBlock, {kind:"list_count"}>, sb: SbClient): Promise<ProspectsBuilder> {
      const min = block.range.min ?? 1;
      const max = block.range.max ?? 999999;
      const { data } = await sb.from("property_stack_counts").select("property_id").gte("stack_count", min).lte("stack_count", max);
      const ids = (data ?? []).map(r => r.property_id);
      return ids.length ? builder.in("id", ids) : builder.in("id", ["__no_match__"]);
    }
