    ```ts
    import { decodeFilters, type FilterBlock } from "@/lib/prospects/filter-schema";
    import { translateLegacyChipParams } from "@/lib/prospects/back-compat-url-translator";

    export function parseProspectsSearch(raw: Record<...>): ParsedProspectsSearch {
      const base = parseTableSearch(raw, /* ... existing args ... */);

      // NEW: parse ?filters= JSON state. If present, take precedence over legacy chips.
      const rawFilters = pickFirst(raw.filters);
      let blockStack: FilterBlock[] = [];
      let filtersSource: "url-state" | "legacy-chips" | "none" = "none";
      if (rawFilters) {
        const decoded = decodeFilters(rawFilters);
        blockStack = decoded.blocks;
        filtersSource = "url-state";
      } else {
        const legacy = translateLegacyChipParams(raw);
        if (legacy.length > 0) {
          blockStack = legacy;
          filtersSource = "legacy-chips";
        }
      }

      return {
        ...base,
        filters: parseProspectsFilters(raw),  // keep for table.tsx legacy chip rendering during transition (deletable in v1.1)
        blockStack,
        filtersSource,
      };
    }
    ```
