    ```tsx
    import { createClient } from "@/lib/supabase/server";
    import { QuickFilterChip, type Preset } from "./quick-filter-chip";
    import type { FilterState } from "@/lib/prospects/filter-schema";

    export default async function QuickFiltersBar({ orgId, currentFilterStateRaw }: { orgId: string; currentFilterStateRaw: string | null }) {
      const sb = await createClient();
      // RLS handles: returns rows where user_id = auth.uid() OR (is_base AND member of org).
      // We narrow to: starred-by-me OR is_base, scoped to this org_id (avoid cross-org base leak in multi-org users).
      const { data, error } = await sb
        .from("saved_filters")
        .select("id, name, filters_json, starred, is_base")
        .eq("org_id", orgId)
        .or("is_base.eq.true,starred.eq.true")
        .order("is_base", { ascending: false })
        .order("name", { ascending: true });

      if (error) {
        console.error("[QuickFiltersBar] failed to load presets:", error.message);
        return <div className="text-xs text-muted-foreground">Failed to load Quick Filters.</div>;
      }
      const presets = (data ?? []) as Preset[];
      if (presets.length === 0) return null;

      return (
        <div className="flex flex-wrap gap-2 items-center" data-quick-filters-bar>
          <span className="text-xs text-muted-foreground mr-1">Quick Filters:</span>
          {presets.map(p => (
            <QuickFilterChip key={p.id} preset={p} orgId={orgId} currentFilterStateRaw={currentFilterStateRaw} />
          ))}
        </div>
      );
    }
    ```

    Step 2 — `quick-filter-chip.tsx` (client):

    ```tsx
    "use client";
    import { useRouter, useSearchParams, usePathname } from "next/navigation";
    import { Badge } from "@/components/ui/badge";
    import { encodeFilters, decodeFilters, type FilterState, type FilterBlock } from "@/lib/prospects/filter-schema";

    export type Preset = { id: string; name: string; filters_json: FilterState; starred: boolean; is_base: boolean };
    export type QuickFilterChipProps = { preset: Preset; orgId: string; currentFilterStateRaw: string | null };

    /** Compare two block stacks ignoring the `id` field (user-saved blocks have UUIDs different from the URL state). */
    function blocksEqualIgnoringId(a: FilterBlock[], b: FilterBlock[]): boolean {
      if (a.length !== b.length) return false;
      const norm = (b: FilterBlock) => { const { id: _id, ...rest } = b as any; return rest; };
      return JSON.stringify(a.map(norm)) === JSON.stringify(b.map(norm));
    }

    export function QuickFilterChip({ preset, currentFilterStateRaw }: QuickFilterChipProps) {
      const router = useRouter();
      const pathname = usePathname();
      const params = useSearchParams();
      const current = decodeFilters(currentFilterStateRaw);
      const isActive = blocksEqualIgnoringId(current.blocks, preset.filters_json.blocks ?? []);

      const onClick = () => {
        const next = new URLSearchParams(params?.toString() ?? "");
        if (isActive) {
          next.delete("filters");
        } else {
          next.set("filters", encodeFilters(preset.filters_json));
        }
        const url = `${pathname}?${next.toString()}`.replace(/\?$/, "");
        router.replace(url, { scroll: false });
        router.refresh();
      };

      return (
        <button
          onClick={onClick}
          aria-pressed={isActive}
          className="focus:outline-none"
          data-quick-filter-chip
          data-preset-name={preset.name}
          data-active={isActive}
        >
          <Badge variant={isActive ? "default" : "outline"} className="cursor-pointer">{preset.name}</Badge>
        </button>
      );
    }
    ```

    ```tsx
    "use client";
    import { XIcon } from "lucide-react";
    import { Badge } from "@/components/ui/badge";
    import { Button } from "@/components/ui/button";
    import { useFilterState } from "./use-filter-state";
    import { BLOCK_PICKER_GROUPS } from "./add-block-picker";
    import type { FilterBlock } from "@/lib/prospects/filter-schema";

    function labelFor(kind: FilterBlock["kind"]): string {
      for (const g of BLOCK_PICKER_GROUPS) for (const it of g.items) if (it.kind === kind) return it.label;
      return kind;
    }

    /** One-line summary of a configured block, displayed inside the chip. */
    function summarize(block: FilterBlock): string {
      switch (block.kind) {
        case "vacancy": case "absentee": case "has_unread_inbound": case "needs_human_attention": case "has_open_tasks":
          return block.tri === "any" ? "" : block.tri === "yes" ? "Yes" : "No";
        case "list": case "tag": case "cass": case "outreach_dispo": case "source": case "state": case "market":
        case "pipeline_status": case "engagement": case "assignee": case "motivation_level":
          if (block.values.length === 0) return "";
          const verb = block.combinator === "all" ? "All of" : block.combinator === "not" ? "Not" : "Any of";
          return `${verb} [${block.values.length}]`;
        case "list_count": case "beds": case "baths": case "year_built": case "estimated_value": case "equity_pct":
          const min = block.range.min, max = block.range.max;
          if (min == null && max == null) return "";
          if (min != null && max != null) return `${min}–${max}`;
          if (min != null) return `≥ ${min}`;
          return `≤ ${max}`;
        case "created_date":
          if (block.date.mode === "since") return `Since ${block.date.days}d`;
          if (block.date.mode === "prior") return `Prior ${block.date.days}d`;
          return `${block.date.from ?? "…"} – ${block.date.to ?? "…"}`;
      }
    }

    export function ActiveFiltersChips() {
      const { blocks, removeBlock, clearAll } = useFilterState();
      if (blocks.length === 0) return null;
      return (
        <div className="flex flex-wrap gap-1.5 items-center" data-active-filters-chips>
          {blocks.map(b => {
            const summary = summarize(b);
            return (
              <Badge key={b.id} variant="secondary" className="gap-1" data-chip-kind={b.kind}>
                <span>{labelFor(b.kind)}{summary ? `: ${summary}` : ""}</span>
                <button onClick={() => removeBlock(b.id)} aria-label={`Remove ${labelFor(b.kind)} filter`} className="hover:opacity-70">
                  <XIcon className="size-3" />
                </button>
              </Badge>
            );
          })}
          <Button variant="ghost" size="sm" onClick={clearAll}>Clear all</Button>
        </div>
      );
    }
    ```

    ```tsx
    "use client";
    import { useState, useTransition } from "react";
    import { Checkbox } from "@/components/ui/checkbox";
    import { Input } from "@/components/ui/input";
    import { Button } from "@/components/ui/button";
    import { toast } from "sonner";
    import { useRouter } from "next/navigation";
    import { createSavedFilter } from "@/app/(dashboard)/properties/_actions/saved-filters";
    import type { BlockStack } from "@/lib/prospects/filter-schema";

    export function SavePresetInline({ orgId, currentBlocks }: { orgId: string; currentBlocks: BlockStack }) {
      const [enabled, setEnabled] = useState(false);
      const [name, setName] = useState("");
      const [pending, startTransition] = useTransition();
      const router = useRouter();

      const onSave = () => {
        const trimmed = name.trim();
        if (!trimmed) { toast.error("Please enter a preset name."); return; }
        startTransition(async () => {
          const result = await createSavedFilter({ orgId, name: trimmed, filtersJson: { v: 1, blocks: currentBlocks } });
          if (result.ok) {
            toast.success(`Saved preset "${trimmed}".`);
            setEnabled(false);
            setName("");
            router.refresh();
          } else {
            toast.error(`Failed to save preset: ${result.error.message}`);
          }
        });
      };

      return (
        <div className="border-t pt-3 mt-3" data-save-preset-inline>
          <label className="flex items-center gap-2 cursor-pointer">
            <Checkbox checked={enabled} onCheckedChange={(v) => setEnabled(!!v)} />
            <span className="text-sm">Save as new preset…</span>
          </label>
          {enabled ? (
            <div className="flex gap-2 mt-2">
              <Input placeholder="Preset name" value={name} onChange={e => setName(e.target.value)} maxLength={100} />
              <Button size="sm" onClick={onSave} disabled={pending || !name.trim()}>{pending ? "Saving…" : "Save"}</Button>
            </div>
          ) : null}
        </div>
      );
    }
    ```

    Step 2 — `preset-dropdown.tsx`:

    ```tsx
    "use client";
    import { useTransition } from "react";
    import { ChevronDownIcon, StarIcon } from "lucide-react";
    import { useRouter } from "next/navigation";
    import { Button } from "@/components/ui/button";
    import {
      DropdownMenu, DropdownMenuTrigger, DropdownMenuContent,
      DropdownMenuItem, DropdownMenuSeparator, DropdownMenuLabel,
    } from "@/components/ui/dropdown-menu";
    import { useFilterState } from "./use-filter-state";
    import { togglePinSavedFilter } from "@/app/(dashboard)/properties/_actions/saved-filters";
    import { newBlockId, type FilterBlock } from "@/lib/prospects/filter-schema";
    import type { Preset } from "./quick-filter-chip";
    import { toast } from "sonner";

    export function PresetDropdown({ orgId, presets }: { orgId: string; presets: Preset[] }) {
      const { replaceStack } = useFilterState();
      const router = useRouter();
      const [, startTransition] = useTransition();

      const base = presets.filter(p => p.is_base);
      const mine = presets.filter(p => !p.is_base);

      const apply = (p: Preset) => {
        // Rehydrate with fresh UUIDs so editing the loaded stack doesn't clobber the saved preset's IDs.
        const fresh: FilterBlock[] = (p.filters_json.blocks ?? []).map(b => ({ ...b, id: newBlockId() })) as FilterBlock[];
        replaceStack(fresh);
      };

      const togglePin = (p: Preset) => {
        startTransition(async () => {
          const result = await togglePinSavedFilter({ orgId, id: p.id, starred: !p.starred });
          if (result.ok) {
            toast.success(result.value.starred ? `Pinned "${p.name}".` : `Unpinned "${p.name}".`);
            router.refresh();
          } else {
            toast.error(`Couldn't update pin: ${result.error.message}`);
          }
        });
      };

      return (
        <DropdownMenu>
          <DropdownMenuTrigger render={<Button variant="outline" size="sm" className="w-full"><span className="flex-1 text-left">Preset…</span><ChevronDownIcon /></Button>} />
          <DropdownMenuContent>
            {base.length > 0 && (
              <>
                <DropdownMenuLabel>Base</DropdownMenuLabel>
                {base.map(p => (
                  <DropdownMenuItem key={p.id} onSelect={() => apply(p)}>{p.name}</DropdownMenuItem>
                ))}
                <DropdownMenuSeparator />
              </>
            )}
            {mine.length > 0 ? (
              <>
                <DropdownMenuLabel>Mine</DropdownMenuLabel>
                {mine.map(p => (
                  <div key={p.id} className="flex items-center justify-between" data-preset-row data-preset-name={p.name}>
                    <DropdownMenuItem onSelect={() => apply(p)} className="flex-1">{p.name}</DropdownMenuItem>
                    <button onClick={() => togglePin(p)} aria-label={p.starred ? `Unpin ${p.name}` : `Pin ${p.name}`} className="px-2">
                      <StarIcon className={p.starred ? "fill-current" : ""} />
                    </button>
                  </div>
                ))}
              </>
            ) : (
              <DropdownMenuItem disabled>No saved presets yet</DropdownMenuItem>
            )}
          </DropdownMenuContent>
        </DropdownMenu>
      );
    }
    ```
