```ts
// Block component props (every block conforms)
export type BlockProps<K extends BlockKind> = {
  block: Extract<FilterBlock, { kind: K }>;
  onChange: (patch: Partial<Extract<FilterBlock, { kind: K }>>) => void;
  onRemove: () => void;
};

// Options context (provided by page.tsx, consumed by blocks)
export type BlockOptions = {
  lists: Array<{ id: string; name: string; color?: string }>;
  tags: Array<{ id: string; name: string; color?: string }>;     // already filtered to category='custom'
  markets: string[];                                              // from counties.market
  states: string[];                                               // 2-letter codes; precompute from properties.state distinct
  assignees: Array<{ id: string; email: string }>;
  sources: string[];                                              // properties.source enum
  pipelineStatuses: string[];                                     // properties.status enum (e.g., ['prospect','lead','contract','closed'])
  motivationLevels: string[];                                     // properties.motivation_level enum
  outreachDispos: string[];                                       // outreach_dispo enum
  cassStatuses: string[];                                         // ['verified','unverified','invalid','ambiguous']
};

// React context for the above
export const BlockOptionsContext = createContext<BlockOptions | null>(null);

// Registry dispatcher
export const BLOCK_REGISTRY: Record<BlockKind, React.LazyExoticComponent<React.ComponentType<BlockProps<any>>>>;
export function renderBlock(block: FilterBlock, onChange: (patch: Partial<FilterBlock>) => void, onRemove: () => void): JSX.Element;
```

    ```tsx
    "use client";
    import { XIcon } from "lucide-react";
    import { Button } from "@/components/ui/button";
    import { createContext, useContext } from "react";

    export type BlockOptions = { /* per <interfaces> */ };
    export const BlockOptionsContext = createContext<BlockOptions | null>(null);
    export function useBlockOptions(): BlockOptions {
      const v = useContext(BlockOptionsContext);
      if (!v) throw new Error("BlockOptionsContext missing — wrap with <BlockOptionsContext.Provider>");
      return v;
    }

    export function BlockShell({ label, onRemove, children }: { label: string; onRemove: () => void; children: React.ReactNode }) {
      return (
        <div data-block-shell>
          <div className="flex items-center justify-between mb-2">
            <span className="text-sm font-medium">{label}</span>
            <Button variant="ghost" size="icon-sm" aria-label={`Remove ${label} filter`} onClick={onRemove}><XIcon /></Button>
          </div>
          <div>{children}</div>
        </div>
      );
    }

    /** Helper: render the All/Any/Do Not Include selector */
    export function CombinatorSelect({ value, onChange }: { value: "all" | "any" | "not"; onChange: (v: "all" | "any" | "not") => void }) {
      return (
        <select className="text-xs border rounded px-1 py-0.5 mr-2" value={value} onChange={e => onChange(e.target.value as any)}>
          <option value="all">All</option>
          <option value="any">Any</option>
          <option value="not">Do Not Include</option>
        </select>
      );
    }
    ```
