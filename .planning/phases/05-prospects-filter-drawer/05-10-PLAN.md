    test.describe("Phase 05 — Prospects Filter Drawer", () => {
      test("golden path: open drawer, add Vacancy=Yes + List Count >= 2, apply, see updated count", async ({ page }) => {
        await page.goto("/properties");
        await expect(page.getByRole("button", { name: /Filters/i })).toBeVisible();

        // Open drawer
        await page.getByRole("button", { name: /Filters/i }).click();
        await expect(page.getByText(/Add a filter to slice your prospects/i)).toBeVisible();

        // Open picker
        await page.getByRole("button", { name: /\\+ Add Filter Block/i }).click();

        // Search & select Vacancy
        const searchInput = page.getByPlaceholder(/Search filters/i);
        await expect(searchInput).toBeFocused();
        await searchInput.fill("vacant");
        await page.getByRole("button", { name: /^Vacancy$/i }).click();

        // Configure Vacancy = Yes
        await page.getByLabel(/Yes/i).first().check();

        // Add List Count
        await page.getByRole("button", { name: /\\+ Add Filter Block/i }).click();
        await page.getByPlaceholder(/Search filters/i).fill("list count");
        await page.getByRole("button", { name: /List Count/i }).click();
        await page.getByPlaceholder(/min/i).first().fill("2");

        // Wait for the count CTA to update (debounce 250ms + count action)
        await page.waitForTimeout(500);
        const showButton = page.getByRole("button", { name: /Show \\d+ prospects/i });
        await expect(showButton).toBeVisible();
        const showText = await showButton.textContent();
        expect(showText).toMatch(/Show \\d+ prospects/);

        // Apply
        await showButton.click();
        // Drawer closes; URL updates
        await expect(page).toHaveURL(/filters=/);

        // Active filters chip bar should show 2 chips
        const activeBar = page.locator("[data-active-filters-chips]");
        await expect(activeBar).toBeVisible();
        await expect(activeBar.locator("[data-chip-kind=vacancy]")).toBeVisible();
        await expect(activeBar.locator("[data-chip-kind=list_count]")).toBeVisible();

        // Screenshot
        await page.screenshot({ path: "docs/design/screenshots/2026-05-07-phase-05-prospects-filter-v1/golden-path-applied.png", fullPage: true });
      });

      test("Quick Filters bar shows 5 base presets on a clean account", async ({ page }) => {
        await page.goto("/properties");
        const bar = page.locator("[data-quick-filters-bar]");
        await expect(bar).toBeVisible();
        for (const name of ["Stacked", "Vacant", "Engaged", "Cold", "High Equity"]) {
          await expect(bar.locator(`[data-preset-name="${name}"]`)).toBeVisible();
        }
        await page.screenshot({ path: "docs/design/screenshots/2026-05-07-phase-05-prospects-filter-v1/quick-filters-bar.png" });
      });

      test("clicking a Quick Filter chip activates and updates URL", async ({ page }) => {
        await page.goto("/properties");
        const vacantChip = page.locator('[data-preset-name="Vacant"]');
        await vacantChip.click();
        await expect(page).toHaveURL(/filters=/);
        await expect(vacantChip).toHaveAttribute("data-active", "true");

        // Click again → deactivates
        await vacantChip.click();
        await expect(page).not.toHaveURL(/filters=/);

        await expect(activeBar.locator("[data-chip-kind=cass]")).toBeVisible();
      });

      test("Sheet drawer width = 440px override applied", async ({ page }) => {
        await page.goto("/properties");
        await page.getByRole("button", { name: /Filters/i }).click();
        const sheet = page.locator('[data-slot="sheet-content"][data-side="right"]');
        const maxWidth = await sheet.evaluate(el => getComputedStyle(el).maxWidth);
        expect(maxWidth).toBe("440px");
      });

      // R9 — select-all-matching with an active filter applies a bulk action across pagination.
      // SPEC line 154: "bulk actions ... operate on the filtered set, including across pagination
      // when 'select all matching' is chosen." The action panel must show the matching count
      // (not the page count); after the action the new list must contain ALL filtered rows.
      test("bulk Add-to-list under select-all-matching with active filter writes to all filtered rows (R9)", async ({ page, request }) => {
        // 1. Apply Vacancy=Yes filter via the drawer
        await page.goto("/properties");
        await page.getByRole("button", { name: /Filters/i }).click();
        await page.getByRole("button", { name: /\\+ Add Filter Block/i }).click();
        await page.getByPlaceholder(/Search filters/i).fill("vacant");
        await page.getByRole("button", { name: /^Vacancy$/i }).click();
        await page.getByLabel(/^Yes/i).first().check();
        await page.waitForTimeout(500);

        // Capture the filtered count from the drawer CTA — this is the authoritative
        // "matching count" the bulk action must operate over.
        const showButton = page.getByRole("button", { name: /Show \\d+ prospects/i });
        const showText = await showButton.textContent();
        const m = showText?.match(/Show (\\d[\\d,]*) prospects/);
        const matchingCountText = m?.[1] ?? "0";
        const matchingCount = parseInt(matchingCountText.replace(/,/g, ""), 10);
        expect(matchingCount).toBeGreaterThan(0);

        await showButton.click();
        await expect(page).toHaveURL(/filters=/);

        // 2. Select-all-on-page → click "Select all M prospects →" banner.
        const headerCheckbox = page.getByTestId("select-all-checkbox");
        if (await headerCheckbox.count() > 0) {
          await headerCheckbox.click();
        } else {
          // Fallback: click the table-header row checkbox.
          await page.locator('thead input[type="checkbox"]').first().click();
        }
        const banner = page.getByTestId("select-all-banner");
        await expect(banner).toBeVisible();
        const selectAllAcross = page.getByTestId("select-all-across-pages");
        // The banner says "Select all M prospects →" — assert M = matchingCount (not the page size).
        const bannerText = await selectAllAcross.textContent();
        expect(bannerText).toContain(matchingCountText);
        await selectAllAcross.click();

        // After clicking, the banner switches to data-mode="all-matching".
        await expect(page.locator('[data-testid="select-all-banner"][data-mode="all-matching"]')).toBeVisible();

        // 3. Open bulk action menu → click Add to list → choose / type a list name.
        const listName = `R9 e2e ${Date.now()}`;
        await page.getByRole("button", { name: /Bulk actions/i }).click();
        await page.getByRole("menuitem", { name: /Add to list/i }).click();
        // Some "Add to list" UIs let the user create a new list inline. If yours uses a typeahead:
        const listInput = page.getByPlaceholder(/list name/i).or(page.getByLabel(/list name/i));
        await listInput.fill(listName);
        await page.getByRole("button", { name: /Create|Apply|Add/i }).click();
